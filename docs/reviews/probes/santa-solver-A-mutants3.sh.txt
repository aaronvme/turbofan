#!/usr/bin/env bash
# Santa-loop reviewer A, re-verification of w2/solver @22ba7ac: my mutants M1-M6 again, now also
# against the fix round's new Dawn files (oracle-moving, forms). No directory junction: the export
# lives in the critic worktree's git-ignored test-results/, so Node resolves packages from the
# critic's own node_modules (same dependencies as the branch; only a script line differs).
#
# Usage (from the critic worktree root):
#   mkdir -p test-results/santaA-mut && git -C ../w2-solver archive 22ba7ac | tar -x -C test-results/santaA-mut
#   cp ../w2-solver/public/mesh/*.bin test-results/santaA-mut/public/mesh/
#   bash docs/reviews/probes/santa-solver-A-mutants3.sh.txt test-results/santaA-mut
# Output: BASELINE first (must pass), then one line per mutant, KILLED or SURVIVED.
set -u
cd "$1" || exit 2
VITEST="node ../../node_modules/vitest/vitest.mjs"
OUT=/tmp/santaA3

vt() { $VITEST run -c vitest.gpu.config.ts "$@"; }

run() {
  local id="$1" file="$2" from="$3" to="$4"
  shift 4
  cp "$file" "$file.orig"
  node -e '
    const fs = require("fs");
    const [f, a, b] = process.argv.slice(1);
    const s = fs.readFileSync(f, "utf8");
    const n = s.split(a).length - 1;
    if (n !== 1) { console.error(`${f}: ${n} matches`); process.exit(3); }
    fs.writeFileSync(f, s.replace(a, b));
  ' "$file" "$from" "$to" || { mv "$file.orig" "$file"; echo "$id BAD-MUTANT"; return; }
  if vt "$@" > "$OUT/mut-$id.log" 2>&1; then
    echo "$id SURVIVED ($*)"
  else
    echo "$id KILLED ($*): $(grep -E 'Tests +[0-9]' "$OUT/mut-$id.log" | tail -1) $(grep -cE 'binding index|is invalid due to a previous error' "$OUT/mut-$id.log") validation lines"
  fi
  mv "$file.orig" "$file"
}

F=src/flow/gpu/wgsl/flux.ts
K=src/flow/gpu/wgsl/kernels.ts
G=tests/gpu

if vt $G/oracle.test.ts $G/forms.test.ts $G/flow-solver.test.ts $G/oracle-moving.test.ts \
  $G/solver-exact.test.ts > "$OUT/baseline.log" 2>&1; then
  echo "BASELINE passed: $(grep -E 'Tests +[0-9]' "$OUT/baseline.log" | tail -1)"
else
  echo "BASELINE FAILED: $(grep -E 'Tests +[0-9]' "$OUT/baseline.log" | tail -1)"
  exit 1
fi

run M1 $F '    - (chi / (cbar * fa)) * (pR - pL));' '    + (chi / (cbar * fa)) * (pR - pL));' \
  $G/oracle.test.ts $G/forms.test.ts
run M2b $K 'a += wk * vec4f(qa.x, qa.y, qa.z, qa.w / geo[CELL_GEO + c].x);' \
  'a += wk * vec4f(qa.x, qa.y, qa.z, qa.w * geo[CELL_GEO + c].x);' $G/flow-solver.test.ts
run M2c $K 'textureStore(outState, g.xy, vec4f(b.x / P.pInf, a.x / P.rhoInf, P.tInf + b.z, b.y));' \
  'textureStore(outState, g.xy, vec4f(a.x / P.rhoInf, b.x / P.pInf, P.tInf + b.z, b.y));' \
  $G/flow-solver.test.ts
run M3 $K 'acc = acc6(acc, vec4f(mInf * flowSx, (mInf * P.uInf) * flowSx, 0.0, 0.0), vec4f(0.0));' \
  'acc = acc6(acc, -vec4f(mInf * flowSx, (mInf * P.uInf) * flowSx, 0.0, 0.0), vec4f(0.0));' \
  $G/oracle-moving.test.ts $G/forms.test.ts $G/oracle.test.ts $G/solver-exact.test.ts
run M4 $K 'let sa = twoSum4(ba, lo[2u * c] + d.a);' 'let sa = twoSum4(ba, d.a);' \
  $G/solver-exact.test.ts
run M5 $F 'let mRef = min(1.0, max(qAvg / cbar, machCut));' 'let mRef = max(qAvg / cbar, machCut);' \
  $G/forms.test.ts $G/oracle-moving.test.ts $G/oracle.test.ts
run M6 $K '                               dpMax / (rho * c2)));' \
  '                               0.0 * dpMax / (rho * c2)));' $G/oracle.test.ts $G/oracle-moving.test.ts

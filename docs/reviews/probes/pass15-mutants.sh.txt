#!/usr/bin/env bash
# Critic pass 15: hand mutants on trial15. Each is one sed on one src file, run against the named
# tests, then restored with git checkout. KILLED = the tests fail. Usage: mutants.sh AB|C
set -u
cd /c/Users/Aaron/projects/turbofan/.claude/worktrees/critic
T=reports/critic15/tmp; mkdir -p $T
run() { # id file sed-expr tests... (-t filter allowed)
  local id=$1 f=$2 e=$3; shift 3
  cp "$f" $T/mut.bak
  sed -i "$e" "$f"
  if cmp -s "$f" $T/mut.bak; then echo "$id: SED DID NOT APPLY"; return; fi
  if npx vitest run "$@" > $T/mut-$id.log 2>&1; then echo "$id: SURVIVED"; else echo "$id: KILLED ($(grep -cE '^\s+(×|FAIL)' $T/mut-$id.log) failing lines)"; fi
  git checkout -- "$f"
}
if [ "${1:-AB}" = AB ]; then
S=src/flow/reference/solver.ts
PT="tests/unit/flow/precondition.test.ts -t global"
run A1 $S 's/farFieldBeta2: Math.min(1, Math.max(mInf2, pre.minMach \* pre.minMach)),/farFieldBeta2: Math.min(1, Math.max(mInf2, (given?.minMach ?? 1) ** 2)),/' $PT
run A2 $S 's/machCut: Math.max(pre.machCut ?? pre.minMach, cut),/machCut: pre.machCut ?? pre.minMach,/' $PT
run A3 $S 's/const pre = tracking ? this.pre : undefined;/const pre = tracking ? this.opts.lowMachPreconditioning : undefined;/' $PT
run A4 $S 's/opts.cfl ?? (pre !== undefined \&\& pre.minMach < 1/opts.cfl ?? (given !== undefined \&\& given.minMach < 1/' $PT
H="tests/unit/cycle/heldRetry.test.ts tests/unit/cycle/heldFrame.test.ts tests/unit/cycle/gasPathHold.test.ts"

run B2 src/cycle/heldRetry.ts 's/if (walk === null \&\& (walkStartedS === null/if (true \&\& (walkStartedS === null/' $H
run B3 src/cycle/model.ts 's/      this.entering = true;/      this.entering = false;/' $H tests/unit/cycle/lostSolveCause.test.ts
run B4 src/cycle/model.ts 's/    if (r.reached !== null) {/    if (false \&\& r.reached !== null) {/' $H
fi
if [ "${1:-AB}" = C ]; then

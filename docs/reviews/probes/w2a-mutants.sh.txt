#!/bin/bash
# Critic pass 13 (w2/app-ui @f6ee8e2): hand mutants on the trial merge. Each mutant is one sed on
# one file; the file's targeted suites run, and the file is restored from a copy. KILLED means a
# suite failed. Run from the repo root of the trial merge: bash w2a-mutants.sh.txt <out.txt>
set -u
out=${1:-reports/critic13/mutants.txt}
: > "$out"
mutant() { # id file sed-expr tests...
  local id=$1 file=$2 expr=$3
  shift 3
  cp "$file" "$file.orig"
  sed -i "$expr" "$file"
  if cmp -s "$file" "$file.orig"; then
    echo "$id NOT APPLIED ($file)" | tee -a "$out"
  elif npx vitest run "$@" > "reports/critic13/mutant-$id.log" 2>&1; then
    echo "$id SURVIVED ($file)" | tee -a "$out"
  else
    echo "$id KILLED ($file): $(grep -cE '^ +(×|FAIL)' "reports/critic13/mutant-$id.log") failing lines" | tee -a "$out"
  fi
  mv "$file.orig" "$file"
}

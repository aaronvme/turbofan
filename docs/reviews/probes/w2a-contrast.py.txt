# Critic pass 13 probe (w2/app-ui @f6ee8e2, W2A-N9): WCAG 2.x contrast of the overlay's text
# tokens (src/ui/global.css :root) on the panel fill rgb(11 14 19 / 0.8), composited over the
# darkest and a mid studio backdrop seen behind the panels in docs/look-dev/ui/*.jpg.
# Run: python -I w2a-contrast.py.txt
def lin(c):
    c = c / 255
    return c / 12.92 if c <= 0.04045 else ((c + 0.055) / 1.055) ** 2.4


def lum(rgb):
    r, g, b = (lin(v) for v in rgb)
    return 0.2126 * r + 0.7152 * g + 0.0722 * b


def ratio(a, b):
    la, lb = sorted((lum(a), lum(b)), reverse=True)
    return (la + 0.05) / (lb + 0.05)


def hexrgb(h):
    return tuple(int(h[i : i + 2], 16) for i in (1, 3, 5))


PANEL = (11, 14, 19)
ALPHA = 0.8
for name, backdrop in [("bg #07090c", (7, 9, 12)), ("studio grey 128", (128, 128, 128))]:
    fill = tuple(round(ALPHA * p + (1 - ALPHA) * s) for p, s in zip(PANEL, backdrop))
    for token, h in [("--fg", "#e6e9ee"), ("--muted", "#9aa3b1"), ("--faint", "#6b7483")]:
        print(f"{token:8s} on the panel over {name:16s} (fill {fill}): {ratio(hexrgb(h), fill):.2f}:1")

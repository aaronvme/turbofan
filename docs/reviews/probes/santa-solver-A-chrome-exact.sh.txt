#!/usr/bin/env bash
# Santa-loop reviewer A, re-verification of w2/solver @22ba7ac, W2S-A-I3: the exactness suite
# (tests/gpu/solver-exact.test.ts: bitwise interfaces at u_inf 0 and 170 m/s, the M 0.5 free
# stream, tile/block-split bitwise independence, the sub-ulp compensated update on both records,
# D4-50's packed binding) run in Chrome instead of Dawn-in-Node. None of it needs the oracle
# shell, so tests/ladder's node:fs does not stand in the way: only three imports change.
#   - solverDevice / shippedMesh: navigator.gpu and the asset fetched from /mesh/ (as
#     tests/browser/flowSolver.test.ts does);
#   - createDawnGpu() -> navigator.gpu (the default-limits device of the D4-50 test);
#   - closedBoxMesh: inlined (tests/ladder/oracle.ts's definition, which imports node:fs).
#
# Usage (from the critic worktree root, with the export of mutants3.sh.txt, unmutated):
#   bash docs/reviews/probes/santa-solver-A-chrome-exact.sh.txt test-results/santaA-mut
set -u
cd "$1" || exit 2
node -e '
  const fs = require("fs");
  let s = fs.readFileSync("tests/gpu/solver-exact.test.ts", "utf8");
  const rep = (a, b) => {
    if (s.split(a).length !== 2) { console.error("no unique match: " + a); process.exit(3); }
    s = s.replace(a, b);
  };
  rep("import { GpuErrorLog, shippedMesh, solverDevice } from \x27./helpers/solver\x27;",
`import { GpuErrorLog } from "../../src/gpu/device";
import { loadMeshAsset, parseMeshManifest } from "../../src/flow/mesh/loader";
import { solverMeshFromFile } from "../../src/flow/reference/solverMesh";
const solverDevice = (log: GpuErrorLog) =>
  requestSimDevice(navigator.gpu, {
    powerPreference: "high-performance",
    optionalFeatures: ["timestamp-query"],
    maxLimits: ["maxStorageBuffersPerShaderStage", "maxComputeWorkgroupStorageSize",
      "maxStorageBufferBindingSize", "maxBufferSize"],
    onError: log.sink,
  });
async function shippedMesh() {
  const manifest = parseMeshManifest(await (await fetch("/mesh/manifest.json")).json());
  const entry = manifest.specs["130k"];
  if (entry === undefined) throw new Error("no 130k mesh");
  const bytes = await (await fetch("/mesh/" + entry.file)).arrayBuffer();
  return solverMeshFromFile((await loadMeshAsset(bytes, entry)).file);
}`);
  rep("import { createDawnGpu } from \x27./helpers/dawn\x27;", "");
  rep("import { closedBoxMesh } from \x27../ladder/oracle\x27;",
`const closedBoxMesh = (p: "f32" | "f64" = "f32") =>
  solverMesh(threeBlockRow(0.2, 0.15, 20, 12, {
    left: BC.SLIP_WALL, right: BC.SLIP_WALL, bottom: BC.AXIS, top: BC.SLIP_WALL }), p);`);
  rep("requestSimDevice(createDawnGpu(),", "requestSimDevice(navigator.gpu,");
  s = s.split("../../src/").join("../../src/").split("../unit/").join("../unit/").split("../../tools/").join("../../tools/");
  fs.writeFileSync("tests/browser/santaA-exact.test.ts", "// GENERATED by santa-solver-A-chrome-exact (review probe)\n" + s);
'
node ../../node_modules/vitest/vitest.mjs run -c vitest.browser.config.ts tests/browser/santaA-exact.test.ts \
  --api.port 5190 --api.strictPort --reporter=verbose

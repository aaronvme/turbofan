// Critic pass 16, w2/flowviz @7dac273 (W2V-I7 / D4-86): checks the ambient floor's colour rationale.
// For each Data-plane quantity: the ambient's place on its colour map, the floor's half-width, and the
// colour difference between the ambient's LUT colour and the LUT colour at the floor's edge(s), as
// CIE76 and CIEDE2000 (sRGB D65 -> CIELAB). Also the legend's floor label for each quantity.
//   node --import ./tools/mesh/ts-resolve.mjs <this file>   (from the checkout's root)
import { pathToFileURL } from 'node:url';
const ROOT = process.env.ROOT ?? '.';
const imp = (p: string) => import(pathToFileURL(`${ROOT}/${p}`).href);
const { bakeLut } = await imp('src/flow/viz/colormaps.ts');
const { FIELD_QUANTITIES, ambientFloor, ambientValue } = await imp('src/flow/viz/fields.ts');
const { legendFor } = await imp('src/flow/viz/legend.ts');

const lin = (c: number) => { const s = c / 255; return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4; };
function lab([r, g, b]: number[]): [number, number, number] {
  const R = lin(r), G = lin(g), B = lin(b);
  const X = (0.4124 * R + 0.3576 * G + 0.1805 * B) / 0.95047;
  const Y = 0.2126 * R + 0.7152 * G + 0.0722 * B;
  const Z = (0.0193 * R + 0.1192 * G + 0.9505 * B) / 1.08883;
  const f = (t: number) => (t > (6 / 29) ** 3 ? Math.cbrt(t) : t / (3 * (6 / 29) ** 2) + 4 / 29);
  return [116 * f(Y) - 16, 500 * (f(X) - f(Y)), 200 * (f(Y) - f(Z))];
}
const de76 = (a: number[], b: number[]) => Math.hypot(a[0] - b[0], a[1] - b[1], a[2] - b[2]);
function de2000(l1: number[], l2: number[]): number {
  const [L1, a1, b1] = l1, [L2, a2, b2] = l2;
  const C1 = Math.hypot(a1, b1), C2 = Math.hypot(a2, b2), Cb = (C1 + C2) / 2;
  const G = 0.5 * (1 - Math.sqrt(Cb ** 7 / (Cb ** 7 + 25 ** 7)));
  const a1p = (1 + G) * a1, a2p = (1 + G) * a2;
  const C1p = Math.hypot(a1p, b1), C2p = Math.hypot(a2p, b2);
  const h = (a: number, b: number) => { const x = (Math.atan2(b, a) * 180) / Math.PI; return x < 0 ? x + 360 : x; };
  const h1p = h(a1p, b1), h2p = h(a2p, b2);
  const dLp = L2 - L1, dCp = C2p - C1p;
  let dhp = h2p - h1p; if (C1p * C2p === 0) dhp = 0; else if (dhp > 180) dhp -= 360; else if (dhp < -180) dhp += 360;
  const dHp = 2 * Math.sqrt(C1p * C2p) * Math.sin((dhp * Math.PI) / 360);
  const Lbp = (L1 + L2) / 2, Cbp = (C1p + C2p) / 2;
  let hbp = h1p + h2p; if (C1p * C2p !== 0) { if (Math.abs(h1p - h2p) > 180) hbp += h1p + h2p < 360 ? 360 : -360; hbp /= 2; }
  const T = 1 - 0.17 * Math.cos(((hbp - 30) * Math.PI) / 180) + 0.24 * Math.cos((2 * hbp * Math.PI) / 180) + 0.32 * Math.cos(((3 * hbp + 6) * Math.PI) / 180) - 0.2 * Math.cos(((4 * hbp - 63) * Math.PI) / 180);
  const SL = 1 + (0.015 * (Lbp - 50) ** 2) / Math.sqrt(20 + (Lbp - 50) ** 2), SC = 1 + 0.045 * Cbp, SH = 1 + 0.015 * Cbp * T;
  const dTh = 30 * Math.exp(-(((hbp - 275) / 25) ** 2)), RC = 2 * Math.sqrt(Cbp ** 7 / (Cbp ** 7 + 25 ** 7)), RT = -RC * Math.sin((2 * dTh * Math.PI) / 180);
  return Math.sqrt((dLp / SL) ** 2 + (dCp / SC) ** 2 + (dHp / SH) ** 2 + RT * (dCp / SC) * (dHp / SH));
}
const REF = { pInfPa: 101325, rhoInfKgPerM3: 1.225, tInfK: 288.15 };
for (const id of Object.keys(FIELD_QUANTITIES)) {
  const s = FIELD_QUANTITIES[id];
  const lut = bakeLut(s.colormap);
  const at = (v: number) => { const u = Math.min(1, Math.max(0, (v - s.min) / (s.max - s.min))); const k = 4 * Math.round(255 * u); return [lut[k], lut[k + 1], lut[k + 2]]; };
  const amb = ambientValue(id, REF), f = ambientFloor(id);
  const c0 = lab(at(amb));
  const edges = [amb - f, amb + f].filter((v) => v >= s.min && v <= s.max);
  const parts = edges.map((v) => { const c = lab(at(v)); return `edge ${v.toFixed(4)}: dE76 ${de76(c0, c).toFixed(2)} dE00 ${de2000(c0, c).toFixed(2)}`; });
  const l = legendFor('data', 'uncoupled', id);
  console.log(`${id} (${s.colormap}, ${s.min}..${s.max} ${s.unit}): ambient ${amb} at u ${((amb - s.min) / (s.max - s.min)).toFixed(3)}, floor +-${f} | ${parts.join(' | ')} | legend: ${l?.floor?.label} [${l?.floor?.from.toFixed(4)}, ${l?.floor?.to.toFixed(4)}]`);
}

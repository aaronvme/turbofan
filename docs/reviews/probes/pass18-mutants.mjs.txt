const want = process.argv[2]?.split(',');
for (const m of M) {
  if (want !== undefined && !want.includes(m.id)) continue;
  const path = join(tree, m.file);
  const orig = readFileSync(path, 'utf8');
  const n = orig.split(m.from).length - 1;
  if (n !== 1) {
    appendFileSync(log, `${m.id} SKIP: the pattern occurs ${n} times in ${m.file}\n`);
    continue;
  }
  writeFileSync(path, orig.replace(m.from, m.to));
  const t0 = Date.now();
  const r = spawnSync('npx', ['vitest', 'run', ...m.tests], { cwd: tree, shell: true, encoding: 'utf8' });
  writeFileSync(path, orig);
  const out = `${r.stdout}\n${r.stderr}`;
  const failed = (out.match(/(\d+) failed/) ?? [])[1] ?? '0';
  const passed = (out.match(/Tests\s+.*?(\d+) passed/) ?? [])[1] ?? '?';
  appendFileSync(log, `${m.id} ${r.status === 0 ? 'SURVIVED' : 'KILLED'} (exit ${String(r.status)}, ${failed} failed, ${passed} passed, ${String(Math.round((Date.now() - t0) / 1000))} s) ${m.file}\n`);
}
appendFileSync(log, `DONE ${new Date().toISOString()}\n`);

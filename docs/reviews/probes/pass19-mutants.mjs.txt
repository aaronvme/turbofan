const set = process.argv[2] ?? 'unit';
const want = process.argv[3]?.split(',');
for (const m of M[set] ?? []) {
  if (want !== undefined && !want.includes(m.id)) continue;
  const path = join(tree, m.file);
  const orig = readFileSync(path, 'utf8');
  const n = orig.split(m.from).length - 1;
  if (n !== 1) {
    appendFileSync(log, `${m.id} SKIP: the pattern occurs ${n} times in ${m.file}\n`);
    continue;
  }
  writeFileSync(path, orig.replace(m.from, m.to));
  const t0 = Date.now();
  const verdicts = [];
  try {
    for (const args of m.runs) {
      const r = spawnSync('npx', args, { cwd: tree, shell: true, encoding: 'utf8', maxBuffer: 1 << 28 });
      const out = `${r.stdout}\n${r.stderr}`;
      writeFileSync(join(here, `mut-${m.id}.log`), out);
      const failed = (out.match(/Tests\s+(\d+) failed/) ?? [])[1] ?? '0';
      const passed = (out.match(/Tests\s+.*?(\d+) passed/) ?? [])[1] ?? '?';
      verdicts.push({ code: r.status, failed, passed });
    }
  } finally {
    writeFileSync(path, orig);
  }
  const killed = verdicts.some((v) => v.code !== 0);
  appendFileSync(
    log,
    `${m.id} ${killed ? 'KILLED' : 'SURVIVED'} (${verdicts.map((v) => `exit ${String(v.code)}, ${v.failed} failed, ${v.passed} passed`).join('; ')}, ${String(Math.round((Date.now() - t0) / 1000))} s) ${m.file}\n`,
  );
}
appendFileSync(log, `DONE ${set} ${new Date().toISOString()}\n`);

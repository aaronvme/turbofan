# ===== rep.mjs =====
// rep.mjs <file> <old> <new>: replaces the one literal occurrence of <old>; exits 2 if not exactly one
import { readFileSync, writeFileSync } from 'node:fs';
const [file, from, to] = process.argv.slice(2);
const s = readFileSync(file, 'utf8');
const n = s.split(from).length - 1;
if (n !== 1) { console.error(`rep: ${n} occurrences in ${file}`); process.exit(2); }
writeFileSync(file, s.replace(from, to));

#!/usr/bin/env bash
# Critic pass 16: GPU and browser hand mutants on trial16 (run alone, nothing else on the GPU).

#     scaling, which would cancel in the residual sum): the tightened flux-scale rows E-L5 / E-L9a must
#     fail it, where D4-48's 1e-3 could not.
# V1: the cut plane's temperature floor about 0 K instead of T_inf (the shader only): the suite's
#     W2V-I7 case reads Mach only, so it should survive; the critic probe's T row shows it.
set -u
cd /c/Users/Aaron/projects/turbofan/.claude/worktrees/critic
D=reports/scratch/c16; T=$D/mut; mkdir -p $T
run() { # id file old new config tests...
  local id=$1 f=$2 old=$3 new=$4 cfg=$5; shift 5
  cp "$f" $T/mut.bak
  if ! node $D/rep.mjs "$f" "$old" "$new"; then echo "$id: DID NOT APPLY"; return; fi
  if npx vitest run -c "$cfg" "$@" > $T/mut-$id.log 2>&1; then echo "$id: SURVIVED"; else echo "$id: KILLED ($(grep -cE '(×|FAIL) ' $T/mut-$id.log) failing lines)"; fi
  cp $T/mut.bak "$f"
}
case "${1:-K1}" in
K1) run K1 src/flow/gpu/wgsl/flux.ts '  let fa4 = vec4f(gm, gm * P.uInf' '  let fa4 = vec4f(gm * (1.0 + 2e-6 * abs(n.y)), gm * P.uInf' vitest.gpu.config.ts tests/gpu/oracle.test.ts ;;
V1) run V1 src/render/flowViz.ts "      const amb = id === 't' ? tInf : float(0);" "      const amb = float(0);" vitest.browser.config.ts tests/browser/flowViz.test.ts -t W2V-I7 ;;
